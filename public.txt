<div class="chat-box">
    <div class="chat-header">Sala de Chat em Tempo Real</div>
    <div class="chat-messages" id="messagesContainer"></div>
    <form class="chat-form" id="chatForm">
        <input type="text" id="userInput" placeholder="O seu nome" required>
        <input type="text" id="msgInput" placeholder="Escreva uma mensagem..." autocomplete="off" required>
        <button type="submit">Enviar</button>
    </form>
</div>

<script>
    const wsProtocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const ws = new WebSocket(`${wsProtocol}//${window.location.host}`);

    const messagesContainer = document.getElementById('messagesContainer');
    const chatForm = document.getElementById('chatForm');
    const userInput = document.getElementById('userInput');
    const msgInput = document.getElementById('msgInput');

    function renderMessage(msg) {
        const div = document.createElement('div');
        const isOwn = userInput.value.trim() !== '' && msg.user === userInput.value.trim();
        div.className = `message ${isOwn ? 'own' : ''}`;

        const header = document.createElement('div');
        header.className = 'message-header';
        
        const userSpan = document.createElement('span');
        userSpan.className = 'message-user';
        userSpan.textContent = msg.user;

        const timeSpan = document.createElement('span');
        timeSpan.textContent = msg.timestamp;

        header.appendChild(userSpan);
        header.appendChild(timeSpan);

        const textDiv = document.createElement('div');
        textDiv.className = 'message-text';
        textDiv.textContent = msg.text;

        div.appendChild(header);
        div.appendChild(textDiv);

        messagesContainer.appendChild(div);
        messagesContainer.scrollTop = messagesContainer.scrollHeight;
    }

    ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        
        if (data.type === 'HISTORY') {
            messagesContainer.innerHTML = '';
            data.payload.forEach(renderMessage);
        } else if (data.type === 'MESSAGE') {
            renderMessage(data.payload);
        }
    };

    chatForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const user = userInput.value.trim();
        const text = msgInput.value.trim();

        if (user && text && ws.readyState === WebSocket.OPEN) {
            ws.send(JSON.stringify({
                type: 'NEW_MESSAGE',
                user: user,
                text: text
            }));
            msgInput.value = '';
            msgInput.focus();
        }
    });
</script>
